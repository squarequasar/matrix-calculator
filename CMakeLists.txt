cmake_minimum_required(VERSION 3.20)
project(matrix_calculator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(matrix src/matrix.cpp)
target_include_directories(matrix PUBLIC include)
target_compile_options(matrix PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(matrix_calculator src/main.cpp)
target_link_libraries(matrix_calculator PRIVATE matrix)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(matrix_tests tests/test_matrix.cpp)
  target_link_libraries(matrix_tests PRIVATE matrix GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(matrix_tests)
endif()
